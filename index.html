function getPatientMode(age, ageUnit){ let days = 0; if(ageUnit === "Days") days = age; else if(ageUnit === "Months") days = age * 30; else if(ageUnit === "Years") days = age * 365; if (days <= 28) { return "આ NICU નું પેશન્ટ છે (≤ 28 Days)"; } else { return "આ PICU નું પેશન્ટ છે (> 28 Days)"; } }
